87struct DiagnosticReport {
88 static constexpr size_t kMaxIssues = 16;
89 std::array<DiagnosticIssue, kMaxIssues> issues{};
90 size_t issue_count = 0;
91 bool pass = true;
92
93 void add(DiagnosticIssue::Severity sev, const char* fmt, ...) noexcept
94 __attribute__((format(printf, 3, 4))) {
95 if (issue_count >= kMaxIssues) return;
96 DiagnosticIssue& it = issues[issue_count++];
97 it.severity = sev;
98 va_list ap; va_start(ap, fmt);
99 vsnprintf(it.message, sizeof(it.message), fmt, ap);
100 va_end(ap);
101 if (sev == DiagnosticIssue::Severity::kError) pass = false;
102 }
103};

113inline DiagnosticReport check(MotorSnapshot const& s,
114 MotorExpectation const& exp) noexcept {
115 DiagnosticReport rep{};
116
117 if (exp.check_commutation_mode) {
118 if (!s.valid_commutation_mode) {
119 rep.add(DiagnosticIssue::Severity::kError,
120 "COMMUTATION_MODE unreadable (chip not responding to NR 70 read)");
121 } else if (s.commutation_mode != exp.expected_commutation_mode) {
122 rep.add(DiagnosticIssue::Severity::kError,
123 "COMMUTATION_MODE=%u, expected %u",
124 static_cast<unsigned>(s.commutation_mode),
125 static_cast<unsigned>(exp.expected_commutation_mode));
126 }
127 }
128
129 if (exp.check_velocity) {
130 const double err = s.actual_velocity_rpm - exp.expected_velocity_rpm;
131 if (std::fabs(err) > exp.velocity_tolerance_rpm) {
132 rep.add(DiagnosticIssue::Severity::kError,
133 "Velocity off-target: actual=%.2f RPM, expected=%.2f \u00B1%.2f RPM (err=%.2f)",
134 s.actual_velocity_rpm, exp.expected_velocity_rpm,
135 exp.velocity_tolerance_rpm, err);
136 }
137 if (exp.check_velocity_direction && exp.expected_velocity_rpm != 0.0) {
138 const bool same_sign = (s.actual_velocity_rpm * exp.expected_velocity_rpm) >= 0.0;
139 if (!same_sign) {
140 rep.add(DiagnosticIssue::Severity::kError,
141 "Velocity direction wrong: actual=%.2f RPM, expected sign matches %.2f RPM",
142 s.actual_velocity_rpm, exp.expected_velocity_rpm);
143 }
144 }
145 }
146
147 if (exp.check_current) {
148 const int abs_i = std::abs(static_cast<int>(s.motor_current_ma));
149 if (abs_i > static_cast<int>(exp.max_motor_current_ma)) {
150 rep.add(DiagnosticIssue::Severity::kError,
151 "Motor current %d mA exceeds max %u mA",
152 abs_i, static_cast<unsigned>(exp.max_motor_current_ma));
153 }
154 }
155
156 if (exp.check_bus_voltage) {
157 if (s.vbus_volts < exp.vbus_min_v || s.vbus_volts > exp.vbus_max_v) {
158 rep.add(DiagnosticIssue::Severity::kError,
159 "Vbus out of range: %.2f V (allowed %.2f \u2026 %.2f)",
160 static_cast<double>(s.vbus_volts),
161 static_cast<double>(exp.vbus_min_v),
162 static_cast<double>(exp.vbus_max_v));
163 }
164 }
165
166 if (exp.check_chip_temperature && s.chip_temp_c > exp.chip_temp_max_c) {
167 rep.add(DiagnosticIssue::Severity::kError,
168 "Chip temperature %.1f \u00B0C exceeds %.1f \u00B0C",
169 static_cast<double>(s.chip_temp_c),
170 static_cast<double>(exp.chip_temp_max_c));
171 }
172
173 if (exp.check_no_gate_driver_errors && s.gate_driver_error_flags != 0) {
174 rep.add(DiagnosticIssue::Severity::kError,
175 "Gate driver error flags non-zero: 0x%08X",
176 static_cast<unsigned>(s.gate_driver_error_flags));
177 }
178
179 const uint32_t masked_general_errors = s.general_error_flags & ~exp.general_error_ignore_mask;
180 if (masked_general_errors != 0) {
181 // Treat un-masked general errors as warnings — many of them are
182 // sticky boot artifacts (config-loaded, watchdog) that the bench
183 // intentionally ignores. Caller can promote with the mask.
184 rep.add(DiagnosticIssue::Severity::kWarn,
185 "GENERAL_ERROR_FLAGS=0x%08X (masked=0x%08X)",
186 static_cast<unsigned>(s.general_error_flags),
187 static_cast<unsigned>(masked_general_errors));
188 }
189
190 return rep;
191}

202template <typename Comm>
203inline DiagnosticReport diagnose_and_log(tmc9660::TMC9660<Comm>& d,
204 MotorContext const& ctx,
205 MotorExpectation const& exp,
206 const char* tag,
207 const char* label = "diag") noexcept {
208 MotorSnapshot s{};
209 (void)d.diagnostics.snapshot(s, ctx);
210 const DiagnosticReport rep = check(s, exp);
211
212 ESP_LOGI(tag, "[%s] === MOTOR SNAPSHOT (pole_pairs=%u, k_RPM=%.2f) ===",
213 label, static_cast<unsigned>(ctx.pole_pairs), ctx.k_rpm());
214
215 ESP_LOGI(tag, "[%s] Bus Vbus=%.2f V T_chip=%.1f \u00B0C",
216 label, static_cast<double>(s.vbus_volts), static_cast<double>(s.chip_temp_c));
217
218 ESP_LOGI(tag, "[%s] Mode COMMUTATION_MODE=%u%s",
219 label, static_cast<unsigned>(s.commutation_mode),
220 s.valid_commutation_mode ? "" : " (READ FAILED)");
221
222 ESP_LOGI(tag,
223 "[%s] Velocity target=%ld (%.2f RPM) actual=%ld (%.2f RPM) ramp=%ld (%.2f RPM)",
224 label,
225 static_cast<long>(s.target_velocity_internal), s.target_velocity_rpm,
226 static_cast<long>(s.actual_velocity_internal), s.actual_velocity_rpm,
227 static_cast<long>(s.ramp_velocity_internal), s.ramp_velocity_rpm);
228
229 ESP_LOGI(tag,
230 "[%s] Position pos=%ld counts (%.3f mech revs, %.1f \u00B0) phi_e=%d (%.1f \u00B0e, %.1f \u00B0mech)",
231 label,
232 static_cast<long>(s.actual_position_counts), s.actual_position_revs,
233 s.actual_position_deg_mech,
234 static_cast<int>(s.phi_e_internal), s.phi_e_deg_elec, s.phi_e_deg_mech);
235
236 ESP_LOGI(tag,
237 "[%s] Currents I_motor=%d mA Iq=%d mA Iuvw=(%d,%d,%d) Uq=%d Uuvw=(%d,%d,%d)",
238 label,
239 static_cast<int>(s.motor_current_ma), static_cast<int>(s.iq_ma),
240 static_cast<int>(s.i_ux), static_cast<int>(s.i_v), static_cast<int>(s.i_wy),
241 static_cast<int>(s.uq),
242 static_cast<int>(s.u_ux), static_cast<int>(s.u_v), static_cast<int>(s.u_wy));
243
244 ESP_LOGI(tag,
245 "[%s] Flags gen_status=0x%08X gen_err=0x%08X gd_err=0x%08X adc_status=0x%08X",
246 label,
247 static_cast<unsigned>(s.general_status_flags),
248 static_cast<unsigned>(s.general_error_flags),
249 static_cast<unsigned>(s.gate_driver_error_flags),
250 static_cast<unsigned>(s.adc_status_flags));
251
252 ESP_LOGI(tag,
253 "[%s] Regulating: %s%s%s",
254 label,
255 s.regulating_velocity ? "VELOCITY " : "",
256 s.regulating_torque ? "TORQUE " : "",
257 s.velocity_reached ? "VEL_REACHED" : "");
258
259 if (rep.issue_count == 0) {
260 ESP_LOGI(tag, "[%s] CHECK PASS (no issues)", label);
261 } else {
262 for (size_t i = 0; i < rep.issue_count; ++i) {
263 const DiagnosticIssue& it = rep.issues[i];
264 switch (it.severity) {
265 case DiagnosticIssue::Severity::kInfo:
266 ESP_LOGI(tag, "[%s] info: %s", label, it.message); break;
267 case DiagnosticIssue::Severity::kWarn:
268 ESP_LOGW(tag, "[%s] warn: %s", label, it.message); break;
269 case DiagnosticIssue::Severity::kError:
270 ESP_LOGE(tag, "[%s] ERROR: %s", label, it.message); break;
271 }
272 }
273 ESP_LOGI(tag, "[%s] CHECK %s (%u issue(s))",
274 label, rep.pass ? "PASS-with-warnings" : "FAIL",
275 static_cast<unsigned>(rep.issue_count));
276 }
277 return rep;
278}
