419 template <typename Func>
420 auto visitDriver(Func&& func, uint8_t deviceIndex = ONBOARD_TMC9660_INDEX) noexcept {
421 // Hold the device lock for the whole visit, so the slot cannot be deleted underneath it;
422 // the handler takes its own lock inside (order: deviceMutex_ -> handler mutex).
423 MutexLockGuard lock(deviceMutex_);
424 Tmc9660Handler* h = (deviceIndex < MAX_TMC9660_DEVICES && deviceActive_[deviceIndex] &&
425 tmcHandlers_[deviceIndex])
426 ? &*tmcHandlers_[deviceIndex]
427 : nullptr;
428 if (h) {
429 return h->visitDriver(std::forward<Func>(func));
430 }
431 // Driver not available - return default
432 using ReturnType = decltype(h->visitDriver(std::forward<Func>(func)));
433 if constexpr (std::is_void_v<ReturnType>) {
434 return;
435 } else {
436 return ReturnType{};
437 }
438 }

702 template <typename Fn>
703 MotorError withReadyDriverLocked(uint8_t device_index, Fn&& fn) noexcept {
704 if (device_index >= MAX_TMC9660_DEVICES || !deviceActive_[device_index] || !tmcHandlers_[device_index]) {
705 return MotorError::DEVICE_NOT_FOUND;
706 }
707 // Never start a bootloader bring-up from an observability call: only drivers that are up.
708 if (!tmcHandlers_[device_index]->IsDriverReady()) {
709 return MotorError::NOT_INITIALIZED;
710 }
711 bool ran = false;
712 tmcHandlers_[device_index]->visitDriver([&](auto& driver) {
713 ran = true;
714 fn(driver);
715 });
716 return ran ? MotorError::SUCCESS : MotorError::NOT_INITIALIZED;
717 }
718
720 void recordBatchLocked(uint8_t device_index, size_t accesses, size_t failed, uint32_t duration_us) noexcept;
721
722 template <typename Fn>
723 bool withSpiDriverLocked(uint8_t device_index, Fn&& fn) noexcept {
724 // A handler that is not up would run its full bootloader init inside visitDriver(); that
725 // belongs to the init paths (PCAL SPI mux first), so only ready drivers are touched here.
726 if (device_index >= MAX_TMC9660_DEVICES || !deviceActive_[device_index] || !tmcHandlers_[device_index] ||
727 !tmcHandlers_[device_index]->IsDriverReady()) {
728 return false;
729 }
730 bool ran = false;
731 tmcHandlers_[device_index]->visitDriver([&](auto& driver) {
732 if constexpr (std::is_same_v<std::decay_t<decltype(driver)>, Tmc9660Handler::SpiDriver>) {
733 ran = true;
734 fn(driver);
735 }
736 });
737 return ran;
738 }
