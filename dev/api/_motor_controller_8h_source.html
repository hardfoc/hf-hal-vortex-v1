335 template <typename Func>
336 auto visitDriver(Func&& func, uint8_t deviceIndex = ONBOARD_TMC9660_INDEX) noexcept {
337 // Hold the device lock for the whole visit, so the slot cannot be deleted underneath it;
338 // the handler takes its own lock inside (order: deviceMutex_ -> handler mutex).
339 MutexLockGuard lock(deviceMutex_);
340 Tmc9660Handler* h = (deviceIndex < MAX_TMC9660_DEVICES && deviceActive_[deviceIndex] &&
341 tmcHandlers_[deviceIndex])
342 ? &*tmcHandlers_[deviceIndex]
343 : nullptr;
344 if (h) {
345 return h->visitDriver(std::forward<Func>(func));
346 }
347 // Driver not available - return default
348 using ReturnType = decltype(h->visitDriver(std::forward<Func>(func)));
349 if constexpr (std::is_void_v<ReturnType>) {
350 return;
351 } else {
352 return ReturnType{};
353 }
354 }

614 template <typename Fn>
615 bool withSpiDriverLocked(uint8_t device_index, Fn&& fn) noexcept {
616 // A handler that is not up would run its full bootloader init inside visitDriver(); that
617 // belongs to the init paths (PCAL SPI mux first), so only ready drivers are touched here.
618 if (device_index >= MAX_TMC9660_DEVICES || !deviceActive_[device_index] || !tmcHandlers_[device_index] ||
619 !tmcHandlers_[device_index]->IsDriverReady()) {
620 return false;
621 }
622 bool ran = false;
623 tmcHandlers_[device_index]->visitDriver([&](auto& driver) {
624 if constexpr (std::is_same_v<std::decay_t<decltype(driver)>, Tmc9660Handler::SpiDriver>) {
625 ran = true;
626 fn(driver);
627 }
628 });
629 return ran;
630 }
