74constexpr const char* CommErrorToString(CommError error) noexcept {
75 switch (error) {
76 case CommError::SUCCESS: return "Success";
77 case CommError::NOT_INITIALIZED: return "Not initialized";
78 case CommError::INITIALIZATION_FAILED: return "Initialization failed";
79 case CommError::BUS_NOT_AVAILABLE: return "Bus not available";
80 case CommError::DEVICE_NOT_FOUND: return "Device not found";
81 case CommError::INVALID_BUS_INDEX: return "Invalid bus index";
82 case CommError::INVALID_DEVICE_INDEX: return "Invalid device index";
83 case CommError::DEVICE_ALREADY_EXISTS: return "Device already exists";
84 case CommError::DEVICE_CREATION_FAILED: return "Device creation failed";
85 case CommError::COMMUNICATION_FAILED: return "Communication failed";
86 case CommError::INVALID_PARAMETER: return "Invalid parameter";
87 case CommError::MUTEX_LOCK_FAILED: return "Mutex lock failed";
88 default: return "Unknown error";
89 }
90}
