75 auto log_fail_gap = [&](uint16_t type, tmcl::ReplyCode st, uint32_t val) {
76 ESP_LOGW(tag,
77 "GAP type=%u (%s) motor=0 TMCL=%s (0x%02X) value=0x%08lX",
78 static_cast<unsigned>(type), tmcl::to_string(static_cast<tmcl::Parameters>(type)),
79 tmcl::to_string(st), static_cast<unsigned>(static_cast<std::uint8_t>(st)),
80 static_cast<unsigned long>(val));
81 };
82 auto log_fail_ggp = [&](uint8_t bank, uint16_t type, tmcl::ReplyCode st, uint32_t val) {
83 ESP_LOGW(tag,
84 "GGP bank=%u type=%u TMCL=%s (0x%02X) value=0x%08lX", static_cast<unsigned>(bank),
85 static_cast<unsigned>(type), tmcl::to_string(st),
86 static_cast<unsigned>(static_cast<std::uint8_t>(st)),
87 static_cast<unsigned long>(val));

90 constexpr uint8_t kGapMotor = 0;
91 unsigned gap_probes = 0;
92 for (uint16_t id = 0; id <= kGapAxisTypeMaxInclusive; ++id) {
93 if (gap_axis_nr_is_table57_gap(id))
94 continue;
95 uint32_t val = 0;
96 tmcl::ReplyCode st = tmcl::ReplyCode::REPLY_OK;
97 const bool xfer = d.sendCommand(tmcl::Op::GAP, id, kGapMotor, 0, &val, &st);
98 (void)xfer;
99 ++n;
100 hist_inc(hist_gap, st);
101 if (st == tmcl::ReplyCode::REPLY_OK || st == tmcl::ReplyCode::REPLY_CMD_LOADED) {
102 gap_ok++;
103 } else {
104 log_fail_gap(id, st, val);
105 }
106 if ((gap_probes++ & 0xFu) == 0)
107 vTaskDelay(1);
108 }
109
110 n = 0;
111 for (uint8_t bank : {static_cast<uint8_t>(0), static_cast<uint8_t>(2), static_cast<uint8_t>(3)}) {
112 uint16_t tmax = (bank == 0) ? kGgpBank0TypeMaxInclusive
113 : (bank == 2) ? kGgpBank2TypeMaxInclusive
114 : kGgpBank3TypeMaxInclusive;
115 uint32_t* hist = (bank == 0) ? hist_ggp0 : (bank == 2) ? hist_ggp2 : hist_ggp3;
116 unsigned* okc = (bank == 0) ? &ggp0_ok : (bank == 2) ? &ggp2_ok : &ggp3_ok;
117
118 for (uint16_t id = 0; id <= tmax; ++id, ++n) {
119 uint32_t val = 0;
120 tmcl::ReplyCode st = tmcl::ReplyCode::REPLY_OK;
121 (void)d.sendCommand(tmcl::Op::GGP, id, bank, 0, &val, &st);
122 hist_inc(hist, st);
123 if (st == tmcl::ReplyCode::REPLY_OK || st == tmcl::ReplyCode::REPLY_CMD_LOADED) {
124 (*okc)++;
125 } else {
126 log_fail_ggp(bank, id, st, val);
127 }
128 if ((n & 0xFu) == 0)
129 vTaskDelay(1);
130 }
131 }

133 auto dump_hist = [&](const char* name, const uint32_t h[256]) {
134 char line[160];
135 size_t pos = 0;
136 pos += static_cast<size_t>(
137 snprintf(line + pos, sizeof(line) - pos, "%s TMCL status counts:", name));
138 for (unsigned i = 0; i < 256; ++i) {
139 if (h[i] == 0)
140 continue;
141 const int left = static_cast<int>(sizeof(line) - pos);
142 if (left < 24)
143 break;
144 pos += static_cast<size_t>(snprintf(line + pos, static_cast<size_t>(left), " [%u]=%lu", i,
145 static_cast<unsigned long>(h[i])));
146 }
147 ESP_LOGI(tag, "%s", line);
148 };
