123constexpr const char* ImuErrorToString(ImuError error) noexcept {
124 switch (error) {
125 case ImuError::SUCCESS: return "Success";
126 case ImuError::NOT_INITIALIZED: return "Not initialized";
127 case ImuError::INITIALIZATION_FAILED: return "Initialization failed";
128 case ImuError::DEVICE_ALREADY_EXISTS: return "Device already exists";
129 case ImuError::DEVICE_NOT_FOUND: return "Device not found";
130 case ImuError::INVALID_DEVICE_INDEX: return "Invalid device index";
131 case ImuError::CANNOT_DELETE_ONBOARD: return "Cannot delete onboard device";
132 case ImuError::INVALID_I2C_ADDRESS: return "Invalid I2C address";
133 case ImuError::DEPENDENCY_NOT_READY: return "Dependency not ready";
134 case ImuError::COMMUNICATION_FAILED: return "Communication failed";
135 case ImuError::I2C_DEVICE_CREATION_FAILED: return "I2C device creation failed";
136 case ImuError::HANDLER_CREATION_FAILED: return "Handler creation failed";
137 case ImuError::INTERRUPT_ERROR: return "Interrupt configuration error";
138 case ImuError::ALREADY_DEINITIALIZED: return "Already deinitialized";
139 case ImuError::MUTEX_LOCK_FAILED: return "Mutex lock failed";
140 default: return "Unknown error";
141 }
142}
