23template <typename D>
24inline void log_probe(const char* tag, const char* case_name, D& d, tmcl::Op op, uint16_t type,
25 uint8_t motor_bank, uint32_t value) noexcept {
26 uint32_t reply = 0;
27 tmcl::ReplyCode st = tmcl::ReplyCode::REPLY_OK;
28 const bool xfer = d.sendCommand(op, type, motor_bank, value, &reply, &st);
29 const bool ok = xfer && (st == tmcl::ReplyCode::REPLY_OK || st == tmcl::ReplyCode::REPLY_CMD_LOADED);
30 ESP_LOGI(tag,
31 "%s %s op=%s type=%u bank/motor=%u value=0x%08lX reply=0x%08lX tmcl=%s %s",
32 ok ? "PASS" : "FAIL", case_name, tmcl::to_string(op), static_cast<unsigned>(type),
33 static_cast<unsigned>(motor_bank), static_cast<unsigned long>(value),
34 static_cast<unsigned long>(reply), tmcl::to_string(st), xfer ? "xfer" : "no-xfer");
35}
