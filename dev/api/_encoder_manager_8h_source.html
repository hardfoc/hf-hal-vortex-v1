57constexpr const char* EncoderErrorToString(EncoderError error) noexcept {
58 switch (error) {
59 case EncoderError::SUCCESS: return "Success";
60 case EncoderError::NOT_INITIALIZED: return "Not initialized";
61 case EncoderError::INITIALIZATION_FAILED: return "Initialization failed";
62 case EncoderError::INVALID_PARAMETER: return "Invalid parameter";
63 case EncoderError::SPI_COMMUNICATION_FAILED: return "SPI communication failed";
64 case EncoderError::CRC_ERROR: return "CRC error";
65 case EncoderError::FRAMING_ERROR: return "Framing error";
66 case EncoderError::SENSOR_ERROR: return "Sensor error";
67 case EncoderError::TIMEOUT: return "Timeout";
68 case EncoderError::MUTEX_LOCK_FAILED: return "Mutex lock failed";
69 default: return "Unknown error";
70 }
