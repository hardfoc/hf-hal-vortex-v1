57constexpr const char* EncoderErrorToString(EncoderError error) noexcept {

220 template <typename Fn>
221 bool VisitDriver(uint8_t deviceIndex, Fn&& fn) noexcept {
222 MutexLockGuard lock(manager_mutex_);
223 As5047uHandler* handler = ReadyHandlerLocked(deviceIndex);
224 if (handler == nullptr) {
225 return false;
226 }
227 bool ran = false;
228 handler->visitDriver([&](as5047u::AS5047U<As5047uSpiAdapter>& driver) {
229 ran = true;
230 fn(driver);
231 });
232 return ran;
233 }
