50inline void dump_pcal95555_lines(const char* tag, const char* label) noexcept {
51 auto& gpio = GpioManager::GetInstance();
52 if (!gpio.EnsureInitialized()) {
53 ESP_LOGW(tag, "[%s] GpioManager not initialised — skipping PCAL dump", label);
54 return;
55 }
56
57 uint16_t bits = 0;
58 auto err = gpio.DebugReadPcal95555InputSnapshot(bits);
59 if (err != hf_gpio_err_t::GPIO_SUCCESS) {
60 ESP_LOGW(tag, "[%s] PCAL95555 input snapshot failed (err=%d)", label,
61 static_cast<int>(err));
62 return;
63 }
64
65 auto bit = [bits](uint8_t pin) -> int { return static_cast<int>((bits >> pin) & 0x1u); };
66
67 ESP_LOGI(tag, "[%s] PCAL95555 raw inputs = 0x%04X (port1<<8 | port0)", label,
68 static_cast<unsigned>(bits));
69 ESP_LOGI(tag,
70 " P0.3 FAULT_STATUS=%d (active-low) P0.4 DRV_EN=%d P0.5 RST_CTRL=%d "
71 "P0.6 PWR_GOOD=%d",
72 bit(3), bit(4), bit(5), bit(6));
73 ESP_LOGI(tag, " P1.5 SPI_COMM_EN=%d (active-low) P1.6 WAKE_CTRL=%d (active-low)",
74 bit(13), bit(14));
75 ESP_LOGI(tag, " P0.1 TMC_GPIO17_EXP_IN=%d P0.2 TMC_GPIO18_EXP_IN=%d", bit(1), bit(2));
76
77 const bool drv_en_high = bit(4) != 0;
78 const bool fault_asserted = bit(3) == 0; // active-low
79 const bool comm_enabled = bit(13) == 0; // active-low
80 const bool wake_high = bit(14) == 0; // active-low (driven low to "wake")
81 ESP_LOGI(tag,
82 " interpretation: DRV_EN %s | FAULT %s | SPI_COMM %s | WAKE %s",
83 drv_en_high ? "ACTIVE (gate driver allowed to switch)" : "INACTIVE",
84 fault_asserted ? "ASSERTED (TMC9660 reports FAULTN low)" : "released",
85 comm_enabled ? "enabled (SPI mux open)" : "disabled",
86 wake_high ? "asserted (chip kept awake)" : "deasserted");
87}

138template <typename Comm>
139inline void dump_params(tmc9660::TMC9660<Comm>& d, const char* tag, const char* label) noexcept {
140 ESP_LOGI(tag, "[%s] -- TMC9660 parameter snapshot --", label);
141 for (const auto& e : kBldcParamTable) {
142 uint32_t value = 0;
143 if (!d.readParameter(e.id, value)) {
144 ESP_LOGW(tag, " %-40s = <GAP failed>", e.name);
145 continue;
146 }
147 if (e.is_signed) {
148 ESP_LOGI(tag, " %-40s = %ld (0x%08X)", e.name,
149 static_cast<long>(static_cast<int32_t>(value)),
150 static_cast<unsigned>(value));
151 } else {
152 ESP_LOGI(tag, " %-40s = %lu (0x%08X)", e.name,
153 static_cast<unsigned long>(value), static_cast<unsigned>(value));
154 }

231template <typename Comm>
232inline void exercise_target_velocity_matrix(tmc9660::TMC9660<Comm>& d, const char* tag,
233 const char* label,
234 bool prime_openloop_voltage = true) noexcept {
235 if (prime_openloop_voltage) {
236 (void)d.torqueFluxControl.setOpenloopVoltage(vortex_bench_safety::kOpenLoopVoltage);
237 vTaskDelay(pdMS_TO_TICKS(vortex_bench_safety::kErratum4TargetCoalesceMs));
238 } else {
239 vTaskDelay(pdMS_TO_TICKS(vortex_bench_safety::kErratum4TargetCoalesceMs));
240 }
241
242 constexpr std::array<VelocityShot, 8> shots = {{
243 {0, false, true, "baseline zero, ramp OFF / DVM ON"},
244 {0, true, true, "baseline zero, ramp ON / DVM ON"},
245 {1, true, true, "tiny +1, ramp ON / DVM ON"},
246 {350, true, true, "small +350, ramp ON / DVM ON"},
247 {-350, true, true, "small -350, ramp ON / DVM ON"},
248 {350, false, true, "small +350, ramp OFF / DVM ON"},
249 {350, true, false, "small +350, ramp ON / DVM OFF (position-based)"},
250 {350, false, false, "small +350, ramp OFF / DVM OFF"},
251 }};
252
253 for (const auto& s : shots) {
254 // Apply prerequisites (best-effort; record whether each succeeded).
255 bool ramp_ok = d.ramp.enable(s.ramp_enable);
256 bool dvm_ok = d.ramp.setDirectVelocityMode(s.direct_velocity_mode);
257 vTaskDelay(pdMS_TO_TICKS(vortex_bench_safety::kAfterRampDvmTweakMs));
258
259 bool tv_ok = d.velocityControl.setTargetVelocityRaw(s.value);
260
261 // Read back the four interesting registers (always — even on failure).
262 uint32_t tv_rb = 0, rv_rb = 0, av_rb = 0, ang_rb = 0, gen_status = 0;
263 bool tv_rb_ok = d.readParameter(tmcl::Parameters::TARGET_VELOCITY, tv_rb);
264 bool rv_rb_ok = d.readParameter(tmcl::Parameters::RAMP_VELOCITY, rv_rb);
265 bool av_rb_ok = d.readParameter(tmcl::Parameters::ACTUAL_VELOCITY, av_rb);
266 bool ang_rb_ok = d.readParameter(tmcl::Parameters::OPENLOOP_ANGLE, ang_rb);
267 (void)d.telemetry.getGeneralStatusFlags(gen_status);
268
269 ESP_LOGI(tag,
270 "[%s] shot %s -> SAP_TV=%s SAP_RAMP=%s SAP_DVM=%s "
271 "TV_rb=%ld RAMP_V_rb=%ld ACTUAL_V_rb=%ld OPENLOOP_ANGLE_rb=%ld "
272 "REG_VEL_bit=%u REG_TORQUE_bit=%u",
273 label, s.note,
274 tv_ok ? "OK" : "FAIL", ramp_ok ? "OK" : "FAIL", dvm_ok ? "OK" : "FAIL",
275 tv_rb_ok ? static_cast<long>(static_cast<int32_t>(tv_rb)) : -1L,
276 rv_rb_ok ? static_cast<long>(static_cast<int32_t>(rv_rb)) : -1L,
277 av_rb_ok ? static_cast<long>(static_cast<int32_t>(av_rb)) : -1L,
278 ang_rb_ok ? static_cast<long>(static_cast<int32_t>(ang_rb)) : -1L,
279 static_cast<unsigned>((gen_status >> 2) & 1u),
280 static_cast<unsigned>((gen_status >> 1) & 1u));
281
282 // Settle a moment so OPENLOOP_ANGLE has a chance to advance if the ramper engaged.
283 vTaskDelay(pdMS_TO_TICKS(150));
284
285 // Tear motion back to zero before next shot, ignoring failures.
286 (void)d.velocityControl.setTargetVelocityRaw(0);
287 vTaskDelay(pdMS_TO_TICKS(vortex_bench_safety::kErratum4TargetCoalesceMs));
288 }

339// ===========================================================================
340// OPENLOOP_ANGLE writability probe + software commutator
341// ===========================================================================
342//
343// In the latest documented Parameter Mode firmware OPENLOOP_ANGLE (id 45) is
344// `R` (read-only) — it's calculated by the chip's internal ramper from
345// TARGET_VELOCITY. But on the chip ROM we have (`051V100`), TARGET_VELOCITY
346// writes are rejected with REPLY_INVALID_VALUE, so the ramper never advances.
347//
348// Some Trinamic firmwares accept SAP writes to "R" parameters anyway. This
349// helper does two things:
350//
351// 1. Tries SAP 45 = {0, 1024, 4096, -4096, 0} and reports REPLY codes +
352// readbacks. If every SAP returns OK and the readback matches, the
353// firmware is effectively letting us drive PHI_E from software.
354//
355// 2. If the writability check passes, runs a `spin_ms` software commutator:
356// writes OPENLOOP_ANGLE = (ang0 + step*tick) mod 32768 every `dt_ms` ms
357// while sampling ACTUAL_TOTAL_MOTOR_CURRENT. A motor with the gate
358// driver alive will drag synchronously with that vector — we'll see the
359// current dance and (if mechanical load allows) the rotor catch up.
360//
361// Caller responsibilities (must be set BEFORE calling):
362// • COMMUTATION_MODE = FOC_OPENLOOP_VOLTAGE_MODE (3)
363// • OPENLOOP_VOLTAGE > 0 (e.g. 600..1200 for a 24 V bus)
364// • DRV_EN asserted
365// • RAMP_ENABLE = 0 (so the chip's ramper doesn't fight us)
366
367template <typename Comm>
368inline bool probe_openloop_angle_writability(tmc9660::TMC9660<Comm>& d,
369 const char* tag) noexcept {
370 ESP_LOGI(tag, "==== OPENLOOP_ANGLE (45) writability probe ====");
371 constexpr std::array<int16_t, 5> shots = {0, 1024, 4096, -4096, 0};
372 bool all_ok = true;
373 for (auto target : shots) {
374 const bool wr = d.writeParameter(tmcl::Parameters::OPENLOOP_ANGLE,
375 static_cast<uint32_t>(target));
376 uint32_t rb = 0;
377 const bool rd = d.readParameter(tmcl::Parameters::OPENLOOP_ANGLE, rb);
378 const int16_t rb_signed = static_cast<int16_t>(rb);
379 ESP_LOGI(tag, " SAP OPENLOOP_ANGLE=%d -> SAP=%s GAP_rb=%d (%s) match=%s",
380 static_cast<int>(target),
381 wr ? "OK" : "FAIL",
382 static_cast<int>(rb_signed),
383 rd ? "OK" : "FAIL",
384 (wr && rd && rb_signed == target) ? "YES" : "no");
385 if (!wr || !rd || rb_signed != target) all_ok = false;
386 vTaskDelay(pdMS_TO_TICKS(20));
387 }
388 ESP_LOGI(tag, " -> OPENLOOP_ANGLE is %sWRITABLE on this firmware",
389 all_ok ? "" : "NOT ");
390 return all_ok;
391}

393template <typename Comm>
394inline void software_commutator_spin(tmc9660::TMC9660<Comm>& d, const char* tag,
395 int16_t step_per_tick = 256,
396 uint32_t dt_ms = 5,
397 uint32_t spin_ms = 4000) noexcept {
398 ESP_LOGI(tag, "==== Software commutator spin: step=%d / %lums for %lums ====",
399 static_cast<int>(step_per_tick),
400 static_cast<unsigned long>(dt_ms),
401 static_cast<unsigned long>(spin_ms));
402 int32_t ang = 0;
403 const uint32_t ticks = (spin_ms + dt_ms - 1) / dt_ms;
404 const uint32_t log_every = (ticks > 50) ? (ticks / 50) : 1;
405 for (uint32_t i = 0; i < ticks; ++i) {
406 ang = static_cast<int16_t>(ang + step_per_tick); // wrap on int16
407 (void)d.writeParameter(tmcl::Parameters::OPENLOOP_ANGLE,
408 static_cast<uint32_t>(static_cast<int16_t>(ang)));
409 if ((i % log_every) == 0) {
410 uint32_t imot = 0, ang_rb = 0;
411 (void)d.readParameter(tmcl::Parameters::ACTUAL_TOTAL_MOTOR_CURRENT, imot);
412 (void)d.readParameter(tmcl::Parameters::OPENLOOP_ANGLE, ang_rb);
413 ESP_LOGI(tag, " t=%lums ang_cmd=%d ang_rb=%d I_motor=%d mA",
414 static_cast<unsigned long>(i * dt_ms),
415 static_cast<int>(static_cast<int16_t>(ang)),
416 static_cast<int>(static_cast<int16_t>(ang_rb)),
417 static_cast<int>(imot));
418 }
419 vTaskDelay(pdMS_TO_TICKS(dt_ms));
420 }
421 ESP_LOGI(tag, " -> commutator loop complete");
422}
